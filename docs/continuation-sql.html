Continuation rate by first category

-- Set these dates to the dashboard's start and end dates (inclusive).
WITH sessions AS (
    SELECT session_id,
           MIN_BY(main_category, click_order) AS first_category,
           COUNT(*) AS clicks
    FROM workspace.default.clickstream_silver
    WHERE event_date BETWEEN DATE '2008-04-01' AND DATE '2008-08-13'
    GROUP BY session_id
)
SELECT CASE first_category
           WHEN 1 THEN 'Trousers'
           WHEN 2 THEN 'Skirts'
           WHEN 3 THEN 'Blouses'
           WHEN 4 THEN 'Sale'
       END AS category,
       COUNT(*) AS starting_sessions,
       COUNT_IF(clicks >= 2) AS continued_sessions,
       ROUND(100.0 * COUNT_IF(clicks >= 2) / COUNT(*), 1) AS continuation_rate_pct
FROM sessions
GROUP BY first_category
ORDER BY first_category;
